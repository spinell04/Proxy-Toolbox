# Number of parallel workers
workers=100

# Default domain for Ping Test (optional)
# Use exactly the format you want to test:
#   google.com          -> TCP connect only
#   http://google.com   -> full HTTP request
#   https://google.com  -> full HTTPS request
domain=google.com

# ─── Exit IP lookups ─────────────────────────────────
# Which address families the IP Uniqueness Test and the Session Monitor ask
# for. Each tool also prompts at runtime, so this is just the default.
#   ipv4  one lookup per check against IPv4-only endpoints (recommended)
#   ipv6  one lookup per check against IPv6-only endpoints
#   both  two lookups per check — double the request volume against free
#         third-party endpoints, and the session monitor's "lookups per
#         minute" figure doubles with it
# A proxy that answers only one family is not broken: in both mode the other
# family is reported as unknown for that check, never as a rotation.
ip_mode=ipv4

# ─── Discord alerts ──────────────────────────────────
# Webhook URL for monitor alerts (leave empty to disable).
# Treat this as a password: anyone holding it can post to your channel.
discord_webhook=

# Consecutive fleet-wide failures before a DOWN alert (Downtime monitor)
discord_down_threshold=3

# Consecutive fleet-wide successes before a RECOVERED alert (Downtime monitor)
discord_up_threshold=2

# ─── Monitor intervals (ms) ──────────────────────────
# Different units, on purpose.
#   monitor_interval_ms  Downtime monitor. Sleeps this long between individual
#                        proxy checks, so with N proxies each one is revisited
#                        every N x interval.
#   session_interval_ms  Session monitor. Checks the whole list in parallel and
#                        sleeps this long between cycles, so every proxy is
#                        checked every interval. Note the cost: each check is a
#                        request to a free third-party IP endpoint, so 100
#                        proxies at 60000 is ~100 lookups a minute, forever.
# Each value is the prompt default, so you can still override it at runtime.
# Blank / invalid / 0 = built-in default (1000 / 60000).
monitor_interval_ms=1000
session_interval_ms=60000

# ─── Save-proxy latency filters (ms) ─────────────────
# When saving filtered proxies to proxyfiles/, keep only proxies under this
# latency. Blank / invalid / 0 = no filter (save all successful proxies).
ping_max_latency_ms=
tm_max_latency_ms=
bayern_max_latency_ms=

# ─── Auto-update ─────────────────────────────────────
# On startup, check GitHub for a newer release, install it and restart.
# The download is verified against the release's SHA-256 before anything is
# replaced, and a failed check never stops the toolbox from starting.
#   on  = stay current automatically (recommended)
#   off = never check; pin whatever binary you have
# Turn it off if you need a specific version, or on a machine running a
# monitor for days that must not restart underneath you.
auto_update=on

# ─── Latency measurement ─────────────────────────────
# Whether a DNS lookup counts toward the reported latency.
#   off  Resolve before timing, then connect to the address. This is what
#        `ping` reports, and what makes two proxies comparable: the lookup is
#        paid once per host, so counting it charges one check for a cost every
#        later check avoided.
#   on   Time the lookup too, as a client resolving on every request would.
# Site Request Test always includes it; see docs/tools/ping-test.md.
measure_dns=off
